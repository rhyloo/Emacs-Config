* Introduction
** Why EMACS?
No sé como empecé con EMACS, recuerdo que con aproximadamente 13 años empecé en linux. Visualicé muchos videos de diferentes distros, incluso borré varias veces mi SO - Desde aquí envío un saludo a mis padres por no "matarme" - En cualquier caso, supongo que el ecosistema Linux me llevó a investigar más sobre el tema. Las conferencias de RMS sobe el proyecto GNU y porqué no la página de la propia FSF me llevaron a este maravilloso editor.

He probado muchos, atom, VScode, notepad++, bloc de notas, sublime e incluso vi/vim, pero me seguiría quedando con este. Me adapto en el trabajo he tenido que utilizar vscode y no hay ningún problema, no voy tan fluido como cuando trabajo en mi entorno, pero lo hago sin problemas. Ahora bien siempre que puedo vuelvo a él porque por alguna razón yo lo encuentro fascinante, no tanto por lo que ofrece ya que hay gente que incluso ha jugado al doom en él, supera eso VIM. Creo que es la personalización, creo que esa es la mejor parte. Me atrevería a decir que me llena por dentro construirme mis propias herramientas desde cero - menos mal que no he bajado a nivel de bit, transistor u átomo ;) - bromas aparte, sí creo que es parte de la naturaleza ingenieril y EMACS te da esa posibilidad.

Hoy en día, con la IA, tenemos muchos problemas para escribir código. El cerebro se acostumbra muy rápido a lo fácil. Intentaré en la medida de lo posible no utilizarlo, esto no lo hago por tener un proyecto más, lo hago porque me gusta, así que si la utilizo al menos indicaré que no es mi obra con algún tag o etiqueta.

(defun my/get-pwd ()
  "Put the current file name (include directory) on the clipboard"
  (interactive)
  (let ((filename (if (equal major-mode 'dired-mode)
                      default-directory
                    (buffer-file-name))))
    (when filename
      (with-temp-buffer
        (insert filename)
        (clipboard-kill-region (point-min) (point-max)))
      (message filename))))

(defun my/create-temp ()
  "Permite crear directorios o archivos en /tmp con completado de estilo counsel/find-file."
  (interactive)
  (let* ((choices '("file" "directory"))
         (type (completing-read "Create (file/directory): " choices nil t))
         ;; read-file-name permite navegar y autocompletar
         (path (read-file-name "Name: " "/tmp/")))

    (if (string-equal type "directory")
        (progn
          (make-directory path t)
          (dired path))
      (find-file path))
    (message "Created %s: %s" type path)))

(defun ar/compile-autoclose (buffer string)
  "Handle compilation window based on success or failure."
  (if (string-match "finished" string)
      ;; SUCCESS CASE
      (progn
        (message "Build finished :)")
        (run-with-timer 1 nil
                        (lambda ()
                          (when-let* ((multi-window (> (count-windows) 1))
                                      (live (buffer-live-p buffer))
                                      (window (get-buffer-window buffer t)))
                            (delete-window window)))))
    ;; FAILURE CASE
    (progn
      (message "Compilation Failed: %s" string)
      (when-let ((window (get-buffer-window buffer t)))
        (select-window window)      ;; This moves your cursor to the window
        (goto-char (point-max)))))) ;; Ensure you are looking at the error output

(defun ar/colorize-compilation-buffer ()
  (let ((inhibit-read-only t))
    (ansi-color-apply-on-region (point-min) (point-max))))

(defun ct/create-proper-compilation-window ()
  "Setup the *compilation* window with custom settings."
  (when (not (get-buffer-window "*compilation*"))
    (save-selected-window
      (save-excursion
        (let* ((w (split-window-vertically))
               (h (window-height w)))
          (select-window w)
          (switch-to-buffer "*compilation*")

          ;; Reduce window height
          (shrink-window (- h compilation-window-height))

          ;; Prevent other buffers from displaying inside
          (set-window-dedicated-p w t) 
          )))))

(defun dired-get-size ()
  (interactive)
  (let ((files (dired-get-marked-files)))
    (with-temp-buffer
      ;; Obtener el nombre del host remoto
      (let ((remote-hostname (shell-command-to-string "hostname")))

        ;; Eliminar posibles saltos de línea al final
        (setq remote-hostname (string-trim remote-hostname))

        ;; Dependiendo del nombre de la máquina, ejecutamos diferentes comandos
        (cond
         ;; Caso 1: Si estamos en la máquina local con nombre "DESKTOP-O45GL2P"
         ((or           (string= remote-hostname "DESKTOP-AGD6PUD") 
                        (string= remote-hostname "thinkpad")) 
          (apply 'call-process "du" nil t nil "-sch" files)
          (message "Output of du: %s" (buffer-string)))

         ;; Caso 2: Si estamos en el servidor remoto "debian"
         ((string= remote-hostname "debian")
          (let ((default-directory (expand-file-name my-remote-path)))
            (let* ((cleaned-files
                    (mapcar (lambda (file)
                              (replace-regexp-in-string my-remote-path-mod "" file))
                            files))
                   (du-output (shell-command-to-string (concat "du -sch " (mapconcat 'identity cleaned-files " ")))))
              ;; Mostrar la salida en el buffer de mensajes
              (message "Output of du: %s" du-output))))

         ;; Caso 3: Si estamos en otra máquina, por ejemplo, "other-server"
         ((string= remote-hostname "other-server")
          (let ((default-directory (expand-file-name "/ssh:user@other-server:/path/to/directory")))
            (let ((du-output (shell-command-to-string "du -sch /path/to/directory")))
              ;; Mostrar la salida en el buffer de mensajes
              (message "Output of du: %s" du-output))))

         ;; Si el nombre del host no coincide con los anteriores
         (t
          (message "No se ha definido un comando para esta máquina.")))))))

(defun my/shell-here (&optional dir)
  "Abrir una shell nombrada según el directorio (local o remoto vía Tramp)."
  (interactive)
  (let* ((dir (or dir default-directory))
         (host (or (file-remote-p dir 'host) "local"))
         (name (format "*shell:%s:%s*"
                       host
                       (file-name-nondirectory
                        (directory-file-name dir)))))
    (let ((default-directory dir))
      ;; get-buffer-create + shell respeta display-buffer-alist
      (shell (get-buffer-create name)))))

(defun my/ssh-hosts-from-config ()
  "Leer los nombres de Host definidos en ~/.ssh/config."
  (let ((file (expand-file-name "~/.ssh/config"))
        (hosts '()))
    (when (file-readable-p file)
      (with-temp-buffer
        (insert-file-contents file)
        (goto-char (point-min))
        (while (re-search-forward "^[[:space:]]*Host[[:space:]]+\\(.+\\)$" nil t)
          (dolist (h (split-string (match-string 1)))
            ;; Ignorar patrones con comodines (Host *)
            (unless (string-match-p "[*?]" h)
              (push h hosts))))))
    (delete-dups (nreverse hosts))))

(defun my/shell-remote (host)
  "Abrir una shell remota en un HOST de ~/.ssh/config, preguntando el path."
  (interactive
   (list (completing-read "Host SSH: " (my/ssh-hosts-from-config))))
  (let ((path (read-string (format "Ruta en %s: " host)
                           (format "/ssh:%s:~/" host))))
    (my/shell-here path)))

(defun my/switch-shell ()
  "Saltar a una shell existente mostrando su directorio ACTUAL."
  (interactive)
  (let* ((shells (cl-remove-if-not
                  (lambda (b) (string-prefix-p "*shell" (buffer-name b)))
                  (buffer-list)))
         ;; Construir "nombre  →  directorio-actual" en tiempo real
         (candidates
          (mapcar (lambda (b)
                    (cons (format "%-30s %s"
                                  (buffer-name b)
                                  (with-current-buffer b default-directory))
                          b))
                  shells)))
    (if candidates
        (let ((choice (completing-read "Shell: " candidates)))
          (pop-to-buffer (cdr (assoc choice candidates))))
      (call-interactively #'my/shell-here))))

  ;; sending to it fails on every update (and the package's own error report

  (advice-add 'markdown-preview--send-preview :around

                (lambda (open-preview)

*** Editar viendo el resultado
El Markdown se edita casi como se ve publicado, al estilo de Obsidian: sin asteriscos ni almohadillas a la vista, títulos de varios tamaños, imágenes, y tablas y diagramas Mermaid dibujados en el propio buffer.

- Los ficheros Markdown se abren en modo lectura: sin números de línea y con tablas y diagramas siempre dibujados. =C-x C-q= pasa a edición y vuelve a lectura.
- En lectura el cursor recorre las tablas fila a fila y se oculta sobre las imágenes, donde parpadeaba como un bloque grande.
- En edición, tablas y diagramas se ven dibujados; al entrar con el cursor aparece su Markdown para editarlo, y al salir se vuelven a dibujar. El fichero no cambia.
- =C-c C-x C-m=: mostrar u ocultar las marcas (=**=, =#=, enlaces...).
- =C-c C-x d=: volver a dibujar todo a mano.
- Los diagramas los dibuja Chrome sin ventana con mermaid.js, descargado una vez a la caché; los ya dibujados se reutilizan.

  "mermaid.js release used to draw diagrams; downloaded once to the cache.")

(defvar my/mermaid-chrome (or (executable-find "google-chrome")
                              (executable-find "chromium")
                              (executable-find "chromium-browser"))
  "Headless browser that draws the diagrams.")

(defun my/mermaid--dir ()
  "Cache directory for mermaid.js and drawn diagrams."
  (let ((dir (no-littering-expand-var-file-name "mermaid/")))
    (make-directory dir t)
    dir))

(defun my/mermaid--js ()
  "Local copy of mermaid.js, downloaded the first time it is needed."
  (let ((file (expand-file-name "mermaid.min.js" (my/mermaid--dir))))
    (unless (file-exists-p file)
      (message "Descargando mermaid.js (solo la primera vez)...")
      (url-copy-file my/mermaid-js-url file t))
    file))

(defun my/mermaid--blocks ()
  "Mermaid blocks in the buffer as a list of (START END SOURCE).
START is the opening fence and END the end of the closing one."
  (save-excursion
    (goto-char (point-min))
    (let (blocks)
      (while (re-search-forward "^```mermaid[ \t]*\n\\(\\(?:.*\n\\)*?\\)```[ \t]*$" nil t)
        (push (list (match-beginning 0) (match-end 0) (match-string-no-properties 1)) blocks))
      (nreverse blocks))))

(defun my/mermaid--svg-file (source)
  "Cache file for the diagram drawn from SOURCE."
  (expand-file-name (concat (secure-hash 'sha1 source) ".svg") (my/mermaid--dir)))

(defun my/mermaid--page (sources)
  "HTML page that draws SOURCES and leaves the SVGs, base64 JSON, in #out.
Labels are drawn as plain SVG text: Emacs's SVG renderer cannot show the
HTML labels mermaid uses by default."
  (concat
   "<!DOCTYPE html><html><head><meta charset='utf-8'>"
   (format "<script src='file://%s'></script></head><body><pre id='out'></pre><script>"
           (my/mermaid--js))
   "mermaid.initialize({startOnLoad:false, theme:'dark', htmlLabels:false,"
   " flowchart:{htmlLabels:false}, securityLevel:'strict'});"
   (format "const sources = %s;" (json-encode (vconcat sources)))
   "(async () => { const svgs = [];"
   " for (let i = 0; i < sources.length; i++) {"
   "  try { const svg = (await mermaid.render('m' + i, sources[i])).svg;"
   ;; Emacs needs a real size (mermaid says width=100%) and kept spaces
   ;; between the words, which it otherwise drops from SVG text.
   "   const box = svg.match(/viewBox=\"[-0-9.]+ [-0-9.]+ ([0-9.]+) ([0-9.]+)\"/);"
   "   svgs.push(svg.replace(/^<svg /, '<svg xml:space=\"preserve\" '"
   "     + (box ? 'width=\"' + box[1] + '\" height=\"' + box[2] + '\" ' : ''))"
   "     .replace(/ width=\"100%\"/, '')); }"
   "  catch (e) { svgs.push(null); } }"
   " const json = JSON.stringify(svgs);"
   " document.getElementById('out').textContent ="
   "   btoa(String.fromCharCode(...new TextEncoder().encode(json))); })();"
   "</script></body></html>"))

(defun my/mermaid--draw (sources done)
  "Draw SOURCES in headless Chrome, save each SVG, then call DONE in Emacs."
  (let* ((page (make-temp-file "mermaid" nil ".html" (my/mermaid--page sources)))
         ;; Own throwaway profile: with the default one, a Chrome you already
         ;; have open can grab the request and the headless one never ends.
         (profile (make-temp-file "mermaid-chrome" t))
         (out (generate-new-buffer " *mermaid-chrome*")))
    (make-process
     :name "mermaid" :buffer out :noquery t :connection-type 'pipe
     :command (list my/mermaid-chrome "--headless=new" "--disable-gpu"
                    (concat "--user-data-dir=" profile)
                    "--allow-file-access-from-files" "--virtual-time-budget=10000"
                    "--dump-dom" (concat "file://" page))
     :sentinel
     (lambda (proc _event)
       (unless (process-live-p proc)
         (let ((svgs (with-current-buffer out
                       (goto-char (point-min))
                       (when (re-search-forward "<pre id=\"out\">\\([A-Za-z0-9+/=]+\\)</pre>" nil t)
                         (json-parse-string
                          (decode-coding-string (base64-decode-string (match-string 1)) 'utf-8)
                          :null-object nil)))))
           (cl-loop for source in sources
                    for svg across (or svgs [])
                    when svg do (with-temp-file (my/mermaid--svg-file source) (insert svg)))
           (kill-buffer out)
           (delete-file page)
           (delete-directory profile t)
           (funcall done)))))))

(defun my/md-live--text-width (win)
  "Pixels available for text in WIN: its body minus the line numbers.
An image wider than this cannot be wrapped and Emacs stops drawing the
rest of the window."
  (- (window-body-width win t)
     (with-selected-window win
       (if (bound-and-true-p display-line-numbers-mode) (line-number-display-width t) 0))))

(defun my/md-live--char-width ()
  "Pixel width of a character in this buffer, with its text scale (C-x C-+).
`frame-char-width' ignores the scale, so after shrinking the text tables
came out with too few columns, and after enlarging it with too many."
  (if-let* ((win (get-buffer-window (current-buffer) t)))
      (window-font-width win)
    (frame-char-width)))

(defun my/mermaid--max-width ()
  "Widest a diagram may be: the text column, or its window if narrower.
Computed from the text width, not the selected window: when a file is
opened the diagrams are placed before the buffer is in its window."
  (let ((text (* (if (bound-and-true-p olivetti-mode) olivetti-body-width fill-column)
                 (my/md-live--char-width)))
        (win (get-buffer-window (current-buffer) t)))
    (round (* 0.95 (if win (min text (my/md-live--text-width win)) text)))))

;;;; Live view: blocks shown drawn, source revealed under the cursor

(defvar-local my/md-live--revealed nil
  "Drawn block whose Markdown is shown because the cursor is inside it.
A block is the list of its overlays: one for a diagram, one per line for
a table.")

(defun my/md-live--put (start end display kind)
  "Draw DISPLAY over START..END, a block of KIND (`table' or `diagram')."
  (let ((ov (make-overlay start end nil t nil)))
    (overlay-put ov 'my/md-live kind)
    (overlay-put ov 'my/md-live-display display)
    (overlay-put ov 'evaporate t)
    ;; The block it belongs to; a table regroups its line overlays.
    (overlay-put ov 'my/md-live-block (list ov))
    (my/md-live--draw ov)
    ov))

(defun my/md-live--draw (ov)
  "Show OV's drawing instead of its Markdown."
  (overlay-put ov 'display (overlay-get ov 'my/md-live-display))
  ;; The block starts with markup that markdown-mode hides, and a display
  ;; on invisible text is never shown.  An overlay value wins over the text
  ;; property; this symbol is not in the invisibility spec, so it shows.
  (overlay-put ov 'invisible 'my/md-live-shown))

(defun my/md-live--reveal (ov)
  "Show OV's Markdown so it can be edited."
  (overlay-put ov 'display nil)
  (overlay-put ov 'invisible nil))

(defun my/md-live--at-point ()
  "Drawn block containing point, if any, as the list of its overlays."
  (when-let* ((ov (seq-find (lambda (o) (and (overlay-get o 'my/md-live)
                                             (<= (overlay-start o) (point) (overlay-end o))))
                            (overlays-in (line-beginning-position) (line-end-position)))))
    (overlay-get ov 'my/md-live-block)))

(defun my/md-live--update ()
  "Reveal the block under the cursor; redraw the one it just left.
A read-only buffer is only being read, so everything stays drawn."
  (let ((here (unless buffer-read-only (my/md-live--at-point))))
    (unless (eq here my/md-live--revealed)
      (when (seq-some #'overlay-buffer my/md-live--revealed)
        (mapc #'my/md-live--draw (seq-filter #'overlay-buffer my/md-live--revealed))
        ;; It may have been edited while revealed.
        (my/md-live-refresh))
      (mapc #'my/md-live--reveal here)
      (setq my/md-live--revealed here))))

(defvar-local my/md-live--cursor-hidden nil
  "Non-nil while the cursor is hidden because it sits on an image.")

(defun my/md-live--cursor ()
  "Keep the cursor usable over drawn blocks.
On an image it is a big blinking box, so it is hidden.  On a drawn table
row it goes to the row's start, where it shows, and line moves become
logical: a visual move inside a row drawn on several lines stays put."
  (let* ((display (get-char-property (point) 'display))
         (on-image (and (consp display) (eq (car display) 'image)))
         (row (seq-find (lambda (o) (and (eq (overlay-get o 'my/md-live) 'table)
                                         (overlay-get o 'display)))
                        (overlays-in (line-beginning-position) (line-end-position)))))
    (cond ((and on-image (not my/md-live--cursor-hidden))
           (setq-local cursor-type nil my/md-live--cursor-hidden t))
          ((and my/md-live--cursor-hidden (not on-image))
           (kill-local-variable 'cursor-type)
           (setq my/md-live--cursor-hidden nil)))
    (if row
        (progn (goto-char (overlay-start row))
               (setq-local line-move-visual nil))
      (kill-local-variable 'line-move-visual))))

(defun my/mermaid--show ()
  "Draw every Mermaid block that has a diagram ready."
  (remove-overlays (point-min) (point-max) 'my/md-live 'diagram)
  (pcase-dolist (`(,start ,end ,source) (my/mermaid--blocks))
    (let ((file (my/mermaid--svg-file source)))
      (when (file-exists-p file)
        (my/md-live--put start end
                         (create-image file 'svg nil
                                       :max-width (my/mermaid--max-width)
                                       :ascent 'center)
                         'diagram)))))

(defun my/mermaid-refresh ()
  "Draw the Mermaid blocks that changed and show all diagrams in the buffer."
  (if (not my/mermaid-chrome)
      (message "Para ver los diagramas hace falta Google Chrome o Chromium")
    (let* ((buf (current-buffer))
           (missing (seq-uniq (seq-remove (lambda (s) (file-exists-p (my/mermaid--svg-file s)))
                                          (mapcar #'caddr (my/mermaid--blocks))))))
      (if (null missing)
          (my/mermaid--show)
        (my/mermaid--draw missing
                          (lambda ()
                            (when (buffer-live-p buf)
                              (with-current-buffer buf
                                (my/mermaid--show)
                                (setq my/md-live--revealed nil)
                                (my/md-live--update)))))))))

;;;; Tables

(defun my/md-table--regions ()
  "Markdown tables as (START . END), whole lines, outside code blocks."
  (save-excursion
    (goto-char (point-min))
    (let (tables)
      (while (re-search-forward "^[ \t]*|" nil t)
        (let ((start (line-beginning-position)))
          (forward-line 1)
          (if (and (looking-at "^[ \t]*|?[ \t]*:?-+:?[ \t]*|")
                   (not (markdown-code-block-at-pos start)))
              (progn
                (while (looking-at "^[ \t]*|") (forward-line 1))
                (push (cons start (1- (point))) tables))
            (goto-char (line-end-position)))))
      (nreverse tables))))

(defun my/md-table--clean (cell)
  "CELL as plain text: no emphasis marks, code ticks or link targets."
  (thread-last cell
               (replace-regexp-in-string "\\[\\([^]]*\\)\\]([^)]*)" "\\1")
               (replace-regexp-in-string "\\*\\*\\|__\\|`" "")
               string-trim))

(defun my/md-table--rows (start end)
  "Rows of the table in START..END as lists of cells, separator dropped."
  (let (rows)
    (dolist (line (split-string (buffer-substring-no-properties start end) "\n" t))
      (unless (string-match-p "^[ \t]*|?[ \t:|-]*-[ \t:|-]*$" line)
        (let* ((l (string-trim line))
               (l (string-remove-suffix "|" (string-remove-prefix "|" l)))
               (l (replace-regexp-in-string "\\\\|" "\x1" l)))
          (push (mapcar (lambda (c) (my/md-table--clean (subst-char-in-string ?\x1 ?| c)))
                        (split-string l "|"))
                rows))))
    (nreverse rows)))

(defun my/md-table--wrap (text width)
  "TEXT broken into lines of at most WIDTH columns, at spaces if possible."
  (let (lines (line ""))
    (dolist (word (split-string text " " t))
      (while (> (string-width word) width)
        (unless (string-empty-p line) (push line lines) (setq line ""))
        (push (truncate-string-to-width word width) lines)
        (setq word (substring word (length (truncate-string-to-width word width)))))
      (cond ((string-empty-p line) (setq line word))
            ((<= (+ (string-width line) 1 (string-width word)) width)
             (setq line (concat line " " word)))
            (t (push line lines) (setq line word))))
    (nreverse (if (string-empty-p line) (or lines '("")) (cons line lines)))))

(defun my/md-table--box-chars ()
  "Box-drawing characters if the font draws them one column wide, else ASCII.
When the font lacks them Emacs borrows them from another font, usually
wider, and the borders no longer line up with the text."
  (let* ((box "─═│┌┬┐╞╪╡├┼┤└┴┘")
         (one (string-pixel-width "x")))
    (if (cl-every (lambda (c) (= (string-pixel-width (string c)) one)) box)
        box
      "-=|+++++++++++++")))

(defun my/md-table--render (rows width)
  "Draw ROWS (first one is the header) with box lines in WIDTH columns.
Columns shrink, widest first, until the table fits; cells wrap inside them.
Returns one drawing per line of the Markdown table: the header with the top
border, the header separator, then each row with the border below it."
  (let* ((ncol (apply #'max (mapcar #'length rows)))
         (rows (mapcar (lambda (r) (append r (make-list (- ncol (length r)) ""))) rows))
         (widths (cl-loop for i below ncol
                          collect (max 1 (apply #'max (mapcar (lambda (r) (string-width (nth i r))) rows)))))
         (budget (- width (1+ (* 3 ncol))))
         (box (my/md-table--box-chars))
         (bar (string (aref box 2))))
    (while (and (> (apply #'+ widths) budget) (> (apply #'max widths) 6))
      (let ((i (cl-position (apply #'max widths) widths)))
        (setf (nth i widths) (1- (nth i widths)))))
    (cl-flet ((border (l m r fill)
                (propertize (concat l (mapconcat (lambda (w) (make-string (+ w 2) fill)) widths m) r "\n")
                            'face 'shadow))
              (row (cells face)
                (let* ((wrapped (cl-mapcar #'my/md-table--wrap cells widths))
                       (height (apply #'max (mapcar #'length wrapped))))
                  (mapconcat
                   (lambda (k)
                     (concat (propertize bar 'face 'shadow)
                             (mapconcat (lambda (i)
                                          (concat " " (propertize (truncate-string-to-width
                                                                   (or (nth k (nth i wrapped)) "")
                                                                   (nth i widths) nil ?\s)
                                                                  'face face)
                                                  " " (propertize bar 'face 'shadow)))
                                        (number-sequence 0 (1- ncol)) "")
                             "\n"))
                   (number-sequence 0 (1- height)) ""))))
      (cl-flet ((c (i) (string (aref box i))))
        (let ((middle (border (c 9) (c 10) (c 11) (aref box 0)))
              (bottom (border (c 12) (c 13) (c 14) (aref box 0))))
          (mapcar (lambda (chunk) (string-trim-right chunk "\n"))
                  (append
                   (list (concat (border (c 3) (c 4) (c 5) (aref box 0))
                                 (row (car rows) 'bold))
                         (if (cdr rows) (border (c 6) (c 7) (c 8) (aref box 1)) bottom))
                   (cl-loop for (r . more) on (cdr rows)
                            collect (concat (row r 'default) (if more middle bottom))))))))))

(defun my/md-table--lines (start end)
  "Lines in START..END as (BEG . END) pairs, without their newlines."
  (save-excursion
    (goto-char start)
    (let (lines)
      (while (< (point) end)
        (push (cons (point) (line-end-position)) lines)
        (forward-line 1))
      (nreverse lines))))

(defun my/md-table--show ()
  "Draw every table in the buffer.
Each Markdown line gets the drawing of its own row, so the cursor and the
scroll go through the table row by row instead of jumping over it."
  (remove-overlays (point-min) (point-max) 'my/md-live 'table)
  (let ((width (1- (/ (my/mermaid--max-width) (my/md-live--char-width)))))
    (pcase-dolist (`(,start . ,end) (my/md-table--regions))
      (let ((rows (my/md-table--rows start end)))
        (when rows
          (let ((chunks (my/md-table--render rows width))
                (lines (my/md-table--lines start end)))
            ;; Unstyled parts of a display string take the face of the text
            ;; they cover, here markdown's table/code face, with another
            ;; font and width; force the normal font on the whole drawing.
            (dolist (chunk chunks)
              (add-face-text-property 0 (length chunk) '(:inherit default) t chunk))
            ;; Lines and rows match unless the table has stray separator
            ;; lines; then the whole table is one drawing.
            (unless (= (length chunks) (length lines))
              (setq chunks (list (string-join chunks "\n"))
                    lines (list (cons start end))))
            (let ((block (cl-mapcar (lambda (line chunk)
                                      (my/md-live--put (car line) (cdr line) chunk 'table))
                                    lines chunks)))
              (dolist (ov block) (overlay-put ov 'my/md-live-block block)))))))))

(defun my/md-live-refresh ()
  "Redraw tables now and diagrams that changed (in the background)."
  (interactive)
  (my/md-table--show)
  (my/mermaid-refresh)
  (setq my/md-live--revealed nil)
  (my/md-live--update))

(defvar-local my/md-live--width nil
  "Window width, in pixels, the drawings were last fitted to.")

(defun my/md-live--fit (&optional window)
  "Refit images, tables and diagrams when WINDOW's width changes.
Runs when the buffer is shown in a window or that window is resized (a side
panel opening, Claude's window, a split), so nothing is ever cut off.
Emacs passes the window; its buffer need not be the current one.  These
hooks run in the middle of redisplay, where swapping images leaves the
window half drawn, so the refit itself runs right after, from a timer."
  (when-let* ((win (if (windowp window) window (get-buffer-window (current-buffer))))
              (width (my/md-live--text-width win))
              (buf (window-buffer win)))
    (unless (eql width (buffer-local-value 'my/md-live--width buf))
      (with-current-buffer buf (setq my/md-live--width width))
      (run-at-time 0 nil #'my/md-live--refit buf width))))

(defun my/md-live--rescale ()
  "Redraw for the new text scale."
  (setq my/md-live--width nil)
  (my/md-live--fit))

(defun my/md-live--refit (buf width)
  "Redraw BUF's images, tables and diagrams for a window WIDTH pixels wide."
  (when (buffer-live-p buf)
    (with-current-buffer buf
      (setq-local markdown-max-image-size (cons (round (* 0.95 width)) 600))
      (markdown-remove-inline-images)
      (markdown-display-inline-images)
      (my/md-table--show)
      (my/mermaid--show)
      (setq my/md-live--revealed nil)
      (my/md-live--update))))

(define-minor-mode my/md-live-mode
  "Show Markdown tables and Mermaid diagrams drawn; edit them in place."
  :lighter nil
  (if my/md-live-mode
      (progn
        (add-hook 'post-command-hook #'my/md-live--update nil t)
        (add-hook 'post-command-hook #'my/md-live--cursor nil t)
        ;; Switching between reading and editing hides or shows the block
        ;; under the cursor right away.
        (add-hook 'read-only-mode-hook #'my/md-live--update nil t)
        (add-hook 'after-save-hook #'my/md-live-refresh nil t)
        (add-hook 'window-size-change-functions #'my/md-live--fit nil t)
        (add-hook 'window-buffer-change-functions #'my/md-live--fit nil t)
        ;; A new text scale changes how many columns fit even when the
        ;; window keeps its width.
        (add-hook 'text-scale-mode-hook #'my/md-live--rescale nil t)
        (local-set-key (kbd "C-c C-x d") #'my/md-live-refresh)
        (my/md-live-refresh))
    (remove-hook 'post-command-hook #'my/md-live--update t)
    (remove-hook 'post-command-hook #'my/md-live--cursor t)
    (kill-local-variable 'cursor-type)
    (kill-local-variable 'line-move-visual)
    (remove-hook 'read-only-mode-hook #'my/md-live--update t)
    (remove-hook 'after-save-hook #'my/md-live-refresh t)
    (remove-hook 'window-size-change-functions #'my/md-live--fit t)
    (remove-hook 'window-buffer-change-functions #'my/md-live--fit t)
    (remove-hook 'text-scale-mode-hook #'my/md-live--rescale t)
    (remove-overlays (point-min) (point-max) 'my/md-live)))

      (when-let* ((_ (featurep 'dired-subtree))
                  (base (my/dired--name-column)))
        (goto-char (point-min))
        (while (not (eobp))
          (when-let* ((ov (dired-subtree--get-ov))
                      (depth (overlay-get ov 'dired-subtree-depth))
                      (name (dired-move-to-filename)))
            (let* ((guides (mapconcat
                            (lambda (level)
                              (propertize "│ " 'face
                                          `(:foreground ,(nth (mod (1- level) (length my/dired-tree-colors))
                                                              my/dired-tree-colors))))
                            (number-sequence 1 depth) ""))
                   (prefix (concat (propertize " " 'display `(space :align-to ,base)) guides))
                   ;; The icon is an after-string at the same spot; two
                   ;; after-strings have no reliable order, so the guides
                   ;; go inside the icon's, kept apart to redo them.
                   (icon (seq-find (lambda (o) (overlay-get o 'nerd-icons-dired-overlay))
                                   (overlays-in (1- name) name)))
                   (guide (or icon (let ((o (make-overlay (1- name) name)))
                                     (overlay-put o 'my/dired-guide-own t)
                                     (overlay-put o 'evaporate t)
                                     o)))
                   (own (or (overlay-get guide 'my/dired-icon)
                            (overlay-get guide 'after-string) "")))
              (overlay-put guide 'my/dired-icon own)
              (overlay-put guide 'my/dired-guide t)
              (overlay-put guide 'after-string (concat prefix (if icon own "")))))
          (forward-line 1))))))

;;;; Folder sizes

(defvar-local my/dired--du nil
  "Running du for the folder of this dired buffer.")

(defconst my/dired--size-re
  "[-dl][-rwxsStT]\\{9\\}[.+@]?[ \t]+[0-9]+[ \t]+[^ \t]+\\([ \t]+[^ \t]+\\)"
  "Permissions, links and owner; group 1 is the blank before the size and the size.")

(defconst my/dired--size-width 5
  "Widest size du -h writes, as in \"1023K\" or \"9,5G\".")

(defun my/dired--fix-size-column ()
  "Leave room in the size column for any du size, so names never move.
The sizes du brings later are wider than the 4K or 3 that ls shows for a
folder; written over a narrow column they pushed the name to the right.
Each new listing (the folder, or a subfolder opened with TAB) is widened
by the same amount on every line, keeping ls's alignment; lines already
done are marked and left alone."
  (save-excursion
    (save-restriction
      (widen)
      (with-silent-modifications
        (let (lines)
          (goto-char (point-min))
          (while (not (eobp))
            (when (and (not (get-text-property (line-beginning-position) 'my/dired-size-room))
                       (re-search-forward my/dired--size-re (line-end-position) t))
              (push (cons (copy-marker (match-beginning 1)) (copy-marker (match-end 1))) lines))
            (forward-line 1))
          (when lines
            (let ((extra (max 0 (- (1+ my/dired--size-width)
                                   (apply #'min (mapcar (lambda (l) (- (cdr l) (car l))) lines))))))
              (dolist (l lines)
                (goto-char (car l))
                (insert (make-string extra ?\s))
                (put-text-property (line-beginning-position) (1+ (line-beginning-position))
                                   'my/dired-size-room t)
                (set-marker (car l) nil)
                (set-marker (cdr l) nil)))))))))

(defun my/dired-dir-sizes (&optional dir)
  "Show, for each folder in DIR, the size of what it holds.
ls gives a folder's own size (4K), which says nothing.  du can take a
while on big folders, so it runs in the background and the sizes appear
when it ends.  DIR defaults to the folder of this buffer; remote folders
are skipped."
  (let ((buf (current-buffer))
        ;; Not `dired-current-directory': after a revert the cursor can be
        ;; inside an opened subfolder, and that is what it returns.
        (dir (expand-file-name (or dir default-directory))))
    (my/dired--fix-size-column)
    (unless (file-remote-p dir)
      (let* ((out (generate-new-buffer " *dired-du*"))
             (proc
              (make-process
               :name "dired-du" :buffer out :noquery t
               ;; Unreadable folders only print errors; their size stays as is.
               :command (list "sh" "-c" "du -h -d 1 -- \"$1\" 2>/dev/null" "sh" dir)
               :sentinel
               (lambda (proc _event)
                 (unless (process-live-p proc)
                   (let ((sizes (make-hash-table :test #'equal)))
                     (with-current-buffer out
                       (goto-char (point-min))
                       (while (re-search-forward "^\\([^\t]+\\)\t\\(.+\\)$" nil t)
                         (puthash (directory-file-name (match-string 2)) (match-string 1) sizes)))
                     (kill-buffer out)
                     (when (buffer-live-p buf)
                       (with-current-buffer buf (my/dired--put-sizes sizes)))))))))
        ;; A new listing of the whole folder replaces the previous run.
        (when (string= dir (expand-file-name default-directory))
          (when (process-live-p my/dired--du) (delete-process my/dired--du))
          (setq my/dired--du proc))))))

(defun my/dired--put-sizes (sizes)
  "Write the folder SIZES (path -> size) in the size column of each folder."
  (save-excursion
    (save-restriction
      (widen)
      (goto-char (point-min))
      (with-silent-modifications
        (while (not (eobp))
          (when-let* ((file (dired-get-filename nil t))
                      (size (gethash (directory-file-name file) sizes)))
            (beginning-of-line)
            ;; Right-aligned in the room `my/dired--fix-size-column' left, so
            ;; the line keeps its length and the name stays where it was.
            (when (re-search-forward my/dired--size-re (line-end-position) t)
              (let ((old (match-string 1)))
                ;; string-trim runs a regexp: keep the size match for replace-match.
                (unless (save-match-data (string= (string-trim old) size))
                  (replace-match (string-pad size (max (length old) (1+ (length size))) nil t)
                                 t t nil 1)))))
          (forward-line 1))))))
